Xây dựng Server TCP/IP với Java

Tổng quan Server TCP/IP là cơ sở của hầu hết các ứng dụng mạng. Trong bài viết này, chúng ta sẽ tìm hiểu cách xây dựng một server đơn giản. Tạo Server Socket import java.io.*; import java.net.*; public class TCPServer { private ServerSocket serverSocket; private int port; public TCPServer(int port) { this.port = port; } public void startServer() throws IOException { serverSocket = new ServerSocket(port); System.out.println("Server bắt đầu lắng nghe trên cổng " + port); while (true) { Socket clientSocket = serverSocket.accept(); System.out.println("Client kết nối từ: " + clientSocket.getInetAddress().getHostAddress()); handleClient(clientSocket); } } private void handleClient(Socket socket) { try { BufferedReader in = new BufferedReader( new InputStreamReader(socket.getInputStream()) ); PrintWriter out = new PrintWriter( socket.getOutputStream(), true ); String message; while ((message = in.readLine()) != null) { System.out.println("Nhận được: " + message); out.println("Echo: " + message); } socket.close(); } catch (IOException e) { e.printStackTrace(); } } public static void main(String[] args) throws IOException { TCPServer server = new TCPServer(5000); server.startServer(); } } Các bước thực hiện Tạo ServerSocket: Lắng nghe trên một cổng cụ thể Chấp nhận kết nối: Đợi các client kết nối Xử lý dữ liệu: Đọc và ghi dữ liệu từ/đến client Đóng kết nối: Giải phóng tài nguyên Lợi ích Cho phép nhiều client kết nối Xử lý dữ liệu theo thời gian thực Xây dựng nền tảng cho các ứng dụng phân tán Kết luận Xây dựng server TCP/IP là bước đầu tiên để hiểu lập trình mạng. Từ đó, bạn có thể mở rộng thành các ứng dụng phức tạp hơn.

Khám phá UDP Protocol trong Java

Giới thiệu UDP UDP là một giao thức không tin cậy nhưng nhanh chóng, không yêu cầu thiết lập kết nối. Nó thích hợp cho các ứng dụng yêu cầu tốc độ hơn độ chính xác. UDP vs TCP Tính chất TCP UDP Tin cậy Có Không Thứ tự dữ liệu Bảo đảm Không bảo đảm Tốc độ Chậm hơn Nhanh hơn Kết nối Yêu cầu Không yêu cầu Ví dụ UDP Sender import java.io.*; import java.net.*; public class UDPSender { public static void main(String[] args) throws IOException { DatagramSocket socket = new DatagramSocket(); InetAddress address = InetAddress.getByName("localhost"); int port = 8888; String message = "Hello UDP Server!"; byte[] data = message.getBytes(); DatagramPacket packet = new DatagramPacket( data, data.length, address, port ); socket.send(packet); System.out.println("Gửi: " + message); socket.close(); } } Ví dụ UDP Receiver import java.io.*; import java.net.*; public class UDPReceiver { public static void main(String[] args) throws IOException { DatagramSocket socket = new DatagramSocket(8888); byte[] receiveData = new byte[1024]; System.out.println("Server UDP lắng nghe trên cổng 8888"); while (true) { DatagramPacket packet = new DatagramPacket( receiveData, receiveData.length ); socket.receive(packet); String message = new String(packet.getData(), 0, packet.getLength()); System.out.println("Nhận được: " + message); System.out.println("Từ: " + packet.getAddress().getHostAddress()); } } } Ứng dụng của UDP Video/Audio streaming: Không cần độ tin cậy tuyệt đối Online games: Cần tốc độ thấp độ trễ DNS queries: Mỏng nhẹ và nhanh IoT applications: Tiết kiệm tài nguyên Kết luận UDP là lựa chọn phù hợp khi bạn cần tốc độ cao và có thể chấp nhận mất một số dữ liệu.

Lập trình đa luồng trong Java cho ứng dụng mạng

Tại sao cần đa luồng? Một server cần xử lý nhiều client cùng lúc. Đa luồng cho phép chúng ta làm điều này hiệu quả. Tạo Thread cho mỗi Client import java.io.*; import java.net.*; public class MultiThreadedServer { private ServerSocket serverSocket; public MultiThreadedServer(int port) throws IOException { serverSocket = new ServerSocket(port); } public void start() { System.out.println("Server khởi động trên cổng " + serverSocket.getLocalPort()); while (true) { try { Socket clientSocket = serverSocket.accept(); // Tạo thread mới cho mỗi client ClientHandler handler = new ClientHandler(clientSocket); new Thread(handler).start(); } catch (IOException e) { e.printStackTrace(); } } } public static void main(String[] args) throws IOException { MultiThreadedServer server = new MultiThreadedServer(5000); server.start(); } } class ClientHandler implements Runnable { private Socket socket; public ClientHandler(Socket socket) { this.socket = socket; } @Override public void run() { try { BufferedReader in = new BufferedReader( new InputStreamReader(socket.getInputStream()) ); PrintWriter out = new PrintWriter( socket.getOutputStream(), true ); String message; while ((message = in.readLine()) != null) { System.out.println("Thread " + Thread.currentThread().getId() + ": " + message); out.println("Xử lý: " + message); } socket.close(); } catch (IOException e) { e.printStackTrace(); } } } Thread Pool với ExecutorService import java.io.*; import java.net.*; import java.util.concurrent.*; public class ThreadPoolServer { private ExecutorService executorService; private ServerSocket serverSocket; public ThreadPoolServer(int port, int threadPoolSize) throws IOException { serverSocket = new ServerSocket(port); executorService = Executors.newFixedThreadPool(threadPoolSize); } public void start() { while (true) { try { Socket clientSocket = serverSocket.accept(); executorService.execute(new ClientHandler(clientSocket)); } catch (IOException e) { e.printStackTrace(); } } } public static void main(String[] args) throws IOException { ThreadPoolServer server = new ThreadPoolServer(5000, 10); server.start(); } } Lợi ích của Thread Pool Quản lý tài nguyên: Giới hạn số thread Hiệu suất cao: Tái sử dụng thread Tránh quá tải: Ngăn tạo quá nhiều thread Dễ dàng quản lý: Centralized control Best Practices Sử dụng Thread Pool thay vì tạo thread mới cho mỗi request Đặt timeout cho các kết nối Xử lý exception một cách cẩn thận Đóng tài nguyên đúng cách Kết luận Đa luồng là chìa khóa để xây dựng server hiệu quả có khả năng xử lý nhiều client cùng lúc.

RESTful API Design với Java Spring Boot

Giới thiệu REST REST (Representational State Transfer) là một kiến trúc phần mềm dựa trên HTTP cho các dịch vụ web. Nguyên tắc REST Stateless: Mỗi request chứa toàn bộ thông tin cần thiết Resource-based: Sử dụng URLs để biểu diễn tài nguyên Standard methods: GET, POST, PUT, DELETE Representation: JSON hoặc XML Thiết lập Spring Boot # Sử dụng Spring Initializr hoặc mvn archetype:generate -DgroupId=com.example -DartifactId=rest-api Model và Repository import javax.persistence.*; import lombok.*; @Entity @Data @NoArgsConstructor @AllArgsConstructor public class Post { @Id @GeneratedValue(strategy = GenerationType.IDENTITY) private Long id; private String title; private String content; private String author; } // Repository import org.springframework.data.jpa.repository.JpaRepository; public interface PostRepository extends JpaRepository<Post, Long> { List<Post> findByAuthor(String author); } Controller RESTful import org.springframework.beans.factory.annotation.Autowired; import org.springframework.http.HttpStatus; import org.springframework.http.ResponseEntity; import org.springframework.web.bind.annotation.*; import java.util.List; @RestController @RequestMapping("/api/posts") @CrossOrigin(origins = "*") public class PostController { @Autowired private PostRepository postRepository; // GET - Lấy tất cả bài viết @GetMapping public ResponseEntity<List<Post>> getAllPosts() { List<Post> posts = postRepository.findAll(); return ResponseEntity.ok(posts); } // GET - Lấy bài viết theo ID @GetMapping("/{id}") public ResponseEntity<Post> getPostById(@PathVariable Long id) { return postRepository.findById(id) .map(ResponseEntity::ok) .orElse(ResponseEntity.notFound().build()); } // POST - Tạo bài viết mới @PostMapping public ResponseEntity<Post> createPost(@RequestBody Post post) { Post savedPost = postRepository.save(post); return ResponseEntity .status(HttpStatus.CREATED) .body(savedPost); } // PUT - Cập nhật bài viết @PutMapping("/{id}") public ResponseEntity<Post> updatePost( @PathVariable Long id, @RequestBody Post postDetails) { return postRepository.findById(id) .map(post -> { post.setTitle(postDetails.getTitle()); post.setContent(postDetails.getContent()); post.setAuthor(postDetails.getAuthor()); return ResponseEntity.ok(postRepository.save(post)); }) .orElse(ResponseEntity.notFound().build()); } // DELETE - Xóa bài viết @DeleteMapping("/{id}") public ResponseEntity<?> deletePost(@PathVariable Long id) { return postRepository.findById(id) .map(post -> { postRepository.delete(post); return ResponseEntity.ok().build(); }) .orElse(ResponseEntity.notFound().build()); } // GET - Tìm bài viết theo tác giả @GetMapping("/author/{author}") public ResponseEntity<List<Post>> getPostsByAuthor(@PathVariable String author) { List<Post> posts = postRepository.findByAuthor(author); return ResponseEntity.ok(posts); } } HTTP Status Codes 200 OK - Yêu cầu thành công 201 Created - Tài nguyên tạo mới 204 No Content - Thành công không có nội dung 400 Bad Request - Yêu cầu không hợp lệ 401 Unauthorized - Chưa xác thực 403 Forbidden - Không có quyền 404 Not Found - Không tìm thấy 500 Internal Server Error - Lỗi server Exception Handling import org.springframework.http.HttpStatus; import org.springframework.web.bind.annotation.*; @RestControllerAdvice public class GlobalExceptionHandler { @ExceptionHandler(ResourceNotFoundException.class) @ResponseStatus(HttpStatus.NOT_FOUND) public ErrorResponse handleNotFound(ResourceNotFoundException ex) { return new ErrorResponse("Không tìm thấy", ex.getMessage()); } @ExceptionHandler(Exception.class) @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR) public ErrorResponse handleGeneral(Exception ex) { return new ErrorResponse("Lỗi server", ex.getMessage()); } } Testing API # GET curl http://localhost:8080/api/posts # POST curl -X POST http://localhost:8080/api/posts \ -H "Content-Type: application/json" \ -d '{"title":"Bài mới","content":"Nội dung","author":"Tác giả"}' # PUT curl -X PUT http://localhost:8080/api/posts/1 \ -H "Content-Type: application/json" \ -d '{"title":"Cập nhật","content":"Nội dung mới","author":"Tác giả"}' # DELETE curl -X DELETE http://localhost:8080/api/posts/1 Kết luận Spring Boot làm cho việc xây dựng RESTful API trở nên đơn giản và hiệu quả. Nó cung cấp tất cả công cụ cần thiết cho phát triển ứng dụng web hiện đại.

Xử lý JSON trong Java - Sử dụng Jackson

Giới thiệu Jackson Jackson là thư viện Java mạnh mẽ cho xử lý JSON. Nó hỗ trợ serialization và deserialization tự động. Thêm Dependency <dependency> <groupId>com.fasterxml.jackson.core</groupId> <artifactId>jackson-databind</artifactId> <version>2.15.0</version> </dependency> Model Class import com.fasterxml.jackson.annotation.JsonProperty; import com.fasterxml.jackson.annotation.JsonIgnore; import lombok.*; @Data @NoArgsConstructor @AllArgsConstructor public class User { @JsonProperty("user_id") private Long id; private String name; private String email; @JsonIgnore private String password; private List<Post> posts; } @Data @NoArgsConstructor @AllArgsConstructor public class Post { private Long id; private String title; private String content; } Serialization (Object → JSON) import com.fasterxml.jackson.databind.ObjectMapper; public class JsonSerializationExample { public static void main(String[] args) throws Exception { ObjectMapper mapper = new ObjectMapper(); // Tạo object User user = new User( 1L, "Nguyễn Văn A", "a@example.com", "secret123", new ArrayList<>() ); // Convert object to JSON string String jsonString = mapper.writeValueAsString(user); System.out.println(jsonString); // Pretty print String prettyJson = mapper.writerWithDefaultPrettyPrinter() .writeValueAsString(user); System.out.println(prettyJson); // Write to file mapper.writeValue(new File("user.json"), user); } } Deserialization (JSON → Object) public class JsonDeserializationExample { public static void main(String[] args) throws Exception { ObjectMapper mapper = new ObjectMapper(); String json = "{" + "\"user_id\": 1," + "\"name\": \"Nguyễn Văn A\"," + "\"email\": \"a@example.com\"" + "}"; // Convert JSON string to object User user = mapper.readValue(json, User.class); System.out.println("ID: " + user.getId()); System.out.println("Name: " + user.getName()); // Read from file User userFromFile = mapper.readValue( new File("user.json"), User.class ); // Read as List String jsonArray = "[" + json + "]"; List<User> users = mapper.readValue( jsonArray, mapper.getTypeFactory().constructCollectionType( List.class, User.class ) ); } } Custom Serialization import com.fasterxml.jackson.core.JsonGenerator; import com.fasterxml.jackson.databind.JsonSerializer; import com.fasterxml.jackson.databind.SerializerProvider; import java.io.IOException; public class CustomUserSerializer extends JsonSerializer<User> { @Override public void serialize(User user, JsonGenerator gen, SerializerProvider provider) throws IOException { gen.writeStartObject(); gen.writeNumberField("id", user.getId()); gen.writeStringField("fullName", user.getName().toUpperCase()); gen.writeStringField("contact", user.getEmail()); gen.writeArrayFieldStart("posts"); for (Post post : user.getPosts()) { gen.writeObject(post); } gen.writeEndArray(); gen.writeEndObject(); } } // Sử dụng @JsonSerialize(using = CustomUserSerializer.class) public class User { // ... } Custom Deserialization import com.fasterxml.jackson.core.JsonParser; import com.fasterxml.jackson.databind.DeserializationContext; import com.fasterxml.jackson.databind.JsonDeserializer; import java.io.IOException; public class CustomUserDeserializer extends JsonDeserializer<User> { @Override public User deserialize(JsonParser parser, DeserializationContext context) throws IOException { User user = new User(); while (parser.nextToken() != null) { String fieldName = parser.getCurrentName(); parser.nextToken(); if ("user_id".equals(fieldName)) { user.setId(parser.getLongValue()); } else if ("name".equals(fieldName)) { user.setName(parser.getText()); } // ... } return user; } } Xử lý lỗi public class JsonErrorHandling { public static void main(String[] args) { ObjectMapper mapper = new ObjectMapper(); String invalidJson = "{\"name\": \"John\", invalid}"; try { User user = mapper.readValue(invalidJson, User.class); } catch (JsonMappingException e) { System.out.println("Lỗi mapping: " + e.getMessage()); } catch (JsonParseException e) { System.out.println("Lỗi parse: " + e.getMessage()); } catch (IOException e) { System.out.println("Lỗi IO: " + e.getMessage()); } } } Best Practices Sử dụng annotations cho kiểm soát chi tiết Xử lý null values một cách cẩn thận Validate JSON input Sử dụng versioning cho API changes Document JSON structure Kết luận Jackson là thư viện tuyệt vời cho xử lý JSON trong Java. Nó làm cho việc làm việc với dữ liệu JSON trở nên đơn giản và an toàn.
