Tạo Server Socket import java.io.*; import java.net.*; public class TCPServer { private ServerSocket serverSocket; private int port; public TCPServer(int port) { this.port = port; } public void startServer() throws IOException { serverSocket = new ServerSocket(port); System.out.println("Server bắt đầu lắng nghe trên cổng " + port); while (true) { Socket clientSocket = serverSocket.accept(); System.out.println("Client kết nối từ: " + clientSocket.getInetAddress().getHostAddress()); handleClient(clientSocket); } } private void handleClient(Socket socket) { try { BufferedReader in = new BufferedReader( new InputStreamReader(socket.getInputStream()) ); PrintWriter out = new PrintWriter( socket.getOutputStream(), true ); String message; while ((message = in.readLine()) != null) { System.out.println("Nhận được: " + message); out.println("Echo: " + message); } socket.close(); } catch (IOException e) { e.printStackTrace(); } } public static void main(String[] args) throws IOException { TCPServer server = new TCPServer(5000); server.startServer(); } } Các bước thực hiện Tạo ServerSocket: Lắng nghe trên một cổng cụ thể Chấp nhận kết nối: Đợi các client kết nối Xử lý dữ liệu: Đọc và ghi dữ liệu từ/đến client Đóng kết nối: Giải phóng tài nguyên Lợi ích Cho phép nhiều client kết nối Xử lý dữ liệu theo thời gian thực Xây dựng nền tảng cho các ứng dụng phân tán Kết luận Xây dựng server TCP/IP là bước đầu tiên để hiểu lập trình mạng. Căn bản về JavaScript Async/Await cho lập trình mạng

Giới thiệu Async/Await Async/Await là cách hiện đại để xử lý các hoạt động bất đồng bộ trong JavaScript, đặc biệt hữu ích cho lập trình mạng. Callbacks vs Promises vs Async/Await Callback Hell function fetchData(url, callback) { fetch(url, (error, data) => { if (error) { fetch(url + '/retry', (retryError, retryData) => { if (retryError) callback(retryError); else callback(null, retryData); }); } else { callback(null, data); } }); } Với Async/Await async function fetchDataWithRetry(url) { try { const response = await fetch(url); return await response.json(); } catch (error) { console.log('Retry fetch...'); const retryResponse = await fetch(url + '/retry'); return await retryResponse.json(); } } Ví dụ thực tế async function getNetworkData() { try { // Gọi API const response = await fetch('https://api.example.com/data'); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const data = await response.json(); console.log('Dữ liệu:', data); return data; } catch (error) { console.error('Lỗi:', error); } } Xử lý nhiều yêu cầu // Serial - Lần lượt async function fetchSerially() { const data1 = await fetch('/api/data1').then(r => r.json()); const data2 = await fetch('/api/data2').then(r => r.json()); return [data1, data2]; } // Parallel - Song song async function fetchParallel() { const [response1, response2] = await Promise.all([ fetch('/api/data1'), fetch('/api/data2') ]); const [data1, data2] = await Promise.all([ response1.json(), response2.json() ]); return [data1, data2]; } Timeout xử lý function withTimeout(promise, ms) { return Promise.race([ promise, new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), ms) ) ]); } async function fetchWithTimeout(url, timeoutMs) { try { const response = await withTimeout(fetch(url), timeoutMs); return await response.json(); } catch (error) { console.error('Lỗi:', error); } } Kết luận Async/Await làm cho mã bất đồng bộ dễ đọc và dễ bảo trì hơn, rất quan trọng cho lập trình mạng hiệu quả.

Xây dựng Server Node.js với Express

Giới thiệu Express Express là một framework Node.js nhẹ nhàng và linh hoạt để xây dựng các ứng dụng web và API. Thiết lập dự án npm init -y npm install express Tạo Server cơ bản const express = require('express'); const app = express(); const PORT = 3000; // Middleware app.use(express.json()); // Routes app.get('/', (req, res) => { res.send('Chào mừng đến Server!'); }); app.get('/api/data', (req, res) => { res.json({ message: 'Đây là dữ liệu API', timestamp: new Date() }); }); app.post('/api/message', (req, res) => { const { text } = req.body; res.json({ received: text, status: 'success' }); }); // Error Handler app.use((err, req, res, next) => { console.error(err.stack); res.status(500).json({ error: 'Có lỗi xảy ra' }); }); // Start Server app.listen(PORT, () => { console.log(`Server chạy trên http://localhost:${PORT}`); }); Routing nâng cao // Route parameters app.get('/api/users/:id', (req, res) => { const userId = req.params.id; res.json({ userId: userId }); }); // Query parameters app.get('/api/search', (req, res) => { const query = req.query.q; res.json({ results: query }); }); // Multiple routes app.route('/api/posts') .get((req, res) => { res.json([ { id: 1, title: 'Post 1' }, { id: 2, title: 'Post 2' } ]); }) .post((req, res) => { res.status(201).json({ message: 'Post tạo thành công' }); }); Middleware // Custom middleware const logger = (req, res, next) => { console.log(`${req.method} ${req.path}`); next(); }; app.use(logger); // Authentication middleware const authenticate = (req, res, next) => { const token = req.headers.authorization; if (!token) { return res.status(401).json({ error: 'Chưa xác thực' }); } next(); }; app.get('/api/protected', authenticate, (req, res) => { res.json({ data: 'Dữ liệu được bảo vệ' }); }); CORS để cho phép cross-origin requests const cors = require('cors'); app.use(cors()); // Hoặc cấu hình chi tiết app.use(cors({ origin: 'http://localhost:3001', credentials: true })); Kết luận Express giúp xây dựng server mạng nhanh chóng và có cấu trúc tốt.
