Xử lý JSON trong Java - Sử dụng Jackson

Hướng dẫn sử dụng thư viện Jackson để serialize và deserialize JSON trong Java.
7 tháng 12, 2024 · Lập trình viên

Giới thiệu Jackson#

Jackson là thư viện Java mạnh mẽ cho xử lý JSON. Nó hỗ trợ serialization và deserialization tự động.

Thêm Dependency#

<dependency>
    <groupId>com.fasterxml.jackson.core</groupId>
    <artifactId>jackson-databind</artifactId>
    <version>2.15.0</version>
</dependency>

Model Class#

import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.annotation.JsonIgnore;
import lombok.*;

@Data
@NoArgsConstructor
@AllArgsConstructor
public class User {
    @JsonProperty("user_id")
    private Long id;
    
    private String name;
    private String email;
    
    @JsonIgnore
    private String password;
    
    private List<Post> posts;
}

@Data
@NoArgsConstructor
@AllArgsConstructor
public class Post {
    private Long id;
    private String title;
    private String content;
}

Serialization (Object → JSON)#

import com.fasterxml.jackson.databind.ObjectMapper;

public class JsonSerializationExample {
    public static void main(String[] args) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        
        // Tạo object
        User user = new User(
            1L,
            "Nguyễn Văn A",
            "a@example.com",
            "secret123",
            new ArrayList<>()
        );
        
        // Convert object to JSON string
        String jsonString = mapper.writeValueAsString(user);
        System.out.println(jsonString);
        
        // Pretty print
        String prettyJson = mapper.writerWithDefaultPrettyPrinter()
            .writeValueAsString(user);
        System.out.println(prettyJson);
        
        // Write to file
        mapper.writeValue(new File("user.json"), user);
    }
}

Deserialization (JSON → Object)#

public class JsonDeserializationExample {
    public static void main(String[] args) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        
        String json = "{" +
            "\"user_id\": 1," +
            "\"name\": \"Nguyễn Văn A\"," +
            "\"email\": \"a@example.com\"" +
            "}";
        
        // Convert JSON string to object
        User user = mapper.readValue(json, User.class);
        System.out.println("ID: " + user.getId());
        System.out.println("Name: " + user.getName());
        
        // Read from file
        User userFromFile = mapper.readValue(
            new File("user.json"), 
            User.class
        );
        
        // Read as List
        String jsonArray = "[" + json + "]";
        List<User> users = mapper.readValue(
            jsonArray,
            mapper.getTypeFactory().constructCollectionType(
                List.class, User.class
            )
        );
    }
}

Custom Serialization#

import com.fasterxml.jackson.core.JsonGenerator;
import com.fasterxml.jackson.databind.JsonSerializer;
import com.fasterxml.jackson.databind.SerializerProvider;
import java.io.IOException;

public class CustomUserSerializer extends JsonSerializer<User> {
    @Override
    public void serialize(User user, JsonGenerator gen, 
            SerializerProvider provider) throws IOException {
        gen.writeStartObject();
        gen.writeNumberField("id", user.getId());
        gen.writeStringField("fullName", user.getName().toUpperCase());
        gen.writeStringField("contact", user.getEmail());
        gen.writeArrayFieldStart("posts");
        for (Post post : user.getPosts()) {
            gen.writeObject(post);
        }
        gen.writeEndArray();
        gen.writeEndObject();
    }
}

// Sử dụng
@JsonSerialize(using = CustomUserSerializer.class)
public class User {
    // ...
}

Custom Deserialization#

import com.fasterxml.jackson.core.JsonParser;
import com.fasterxml.jackson.databind.DeserializationContext;
import com.fasterxml.jackson.databind.JsonDeserializer;
import java.io.IOException;

public class CustomUserDeserializer extends JsonDeserializer<User> {
    @Override
    public User deserialize(JsonParser parser,
            DeserializationContext context) throws IOException {
        User user = new User();
        while (parser.nextToken() != null) {
            String fieldName = parser.getCurrentName();
            parser.nextToken();
            
            if ("user_id".equals(fieldName)) {
                user.setId(parser.getLongValue());
            } else if ("name".equals(fieldName)) {
                user.setName(parser.getText());
            }
            // ...
        }
        return user;
    }
}

Xử lý lỗi#

public class JsonErrorHandling {
    public static void main(String[] args) {
        ObjectMapper mapper = new ObjectMapper();
        
        String invalidJson = "{\"name\": \"John\", invalid}";
        
        try {
            User user = mapper.readValue(invalidJson, User.class);
        } catch (JsonMappingException e) {
            System.out.println("Lỗi mapping: " + e.getMessage());
        } catch (JsonParseException e) {
            System.out.println("Lỗi parse: " + e.getMessage());
        } catch (IOException e) {
            System.out.println("Lỗi IO: " + e.getMessage());
        }
    }
}

Best Practices#

  1. Sử dụng annotations cho kiểm soát chi tiết
  2. Xử lý null values một cách cẩn thận
  3. Validate JSON input
  4. Sử dụng versioning cho API changes
  5. Document JSON structure

Kết luận#

Jackson là thư viện tuyệt vời cho xử lý JSON trong Java. Nó làm cho việc làm việc với dữ liệu JSON trở nên đơn giản và an toàn.
