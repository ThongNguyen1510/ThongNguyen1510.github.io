Xây dựng Server Node.js với Express

Hướng dẫn tạo một server mạng đơn giản sử dụng Node.js và Express framework.
10 tháng 12, 2024 · Lập trình viên

Giới thiệu Express#

Express là một framework Node.js nhẹ nhàng và linh hoạt để xây dựng các ứng dụng web và API.

Thiết lập dự án#

npm init -y
npm install express

Tạo Server cơ bản#

const express = require('express');
const app = express();
const PORT = 3000;

// Middleware
app.use(express.json());

// Routes
app.get('/', (req, res) => {
    res.send('Chào mừng đến Server!');
});

app.get('/api/data', (req, res) => {
    res.json({ 
        message: 'Đây là dữ liệu API',
        timestamp: new Date()
    });
});

app.post('/api/message', (req, res) => {
    const { text } = req.body;
    res.json({ 
        received: text,
        status: 'success'
    });
});

// Error Handler
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ 
        error: 'Có lỗi xảy ra' 
    });
});

// Start Server
app.listen(PORT, () => {
    console.log(`Server chạy trên http://localhost:${PORT}`);
});

Routing nâng cao#

// Route parameters
app.get('/api/users/:id', (req, res) => {
    const userId = req.params.id;
    res.json({ userId: userId });
});

// Query parameters
app.get('/api/search', (req, res) => {
    const query = req.query.q;
    res.json({ results: query });
});

// Multiple routes
app.route('/api/posts')
    .get((req, res) => {
        res.json([
            { id: 1, title: 'Post 1' },
            { id: 2, title: 'Post 2' }
        ]);
    })
    .post((req, res) => {
        res.status(201).json({ 
            message: 'Post tạo thành công' 
        });
    });

Middleware#

// Custom middleware
const logger = (req, res, next) => {
    console.log(`${req.method} ${req.path}`);
    next();
};

app.use(logger);

// Authentication middleware
const authenticate = (req, res, next) => {
    const token = req.headers.authorization;
    if (!token) {
        return res.status(401).json({ error: 'Chưa xác thực' });
    }
    next();
};

app.get('/api/protected', authenticate, (req, res) => {
    res.json({ data: 'Dữ liệu được bảo vệ' });
});

CORS để cho phép cross-origin requests#

const cors = require('cors');
app.use(cors());

// Hoặc cấu hình chi tiết
app.use(cors({
    origin: 'http://localhost:3001',
    credentials: true
}));

Kết luận#

Express giúp xây dựng server mạng nhanh chóng và có cấu trúc tốt. Nó rất phù hợp cho các API REST và ứng dụng web.
