Fetch API và HTTP Requests trong JavaScript

Hướng dẫn sử dụng Fetch API để gọi các HTTP requests từ JavaScript.
6 tháng 12, 2024 · Lập trình viên

Giới thiệu Fetch API#

Fetch API cung cấp một cách hiện đại để gọi các HTTP requests từ JavaScript, thay thế XMLHttpRequest.

GET Request#

// Đơn giản
fetch('https://api.example.com/users')
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error('Lỗi:', error));

// Với Async/Await
async function getUsers() {
    try {
        const response = await fetch('https://api.example.com/users');
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        const data = await response.json();
        console.log(data);
    } catch (error) {
        console.error('Lỗi:', error);
    }
}

getUsers();

POST Request#

async function createUser(user) {
    try {
        const response = await fetch('https://api.example.com/users', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(user)
        });
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const result = await response.json();
        console.log('Người dùng tạo:', result);
    } catch (error) {
        console.error('Lỗi:', error);
    }
}

createUser({
    name: 'Nguyễn Văn A',
    email: 'a@example.com'
});

PUT/PATCH Request#

// PUT - Cập nhật toàn bộ
async function updateUser(id, updatedData) {
    try {
        const response = await fetch(
            `https://api.example.com/users/${id}`,
            {
                method: 'PUT',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(updatedData)
            }
        );
        return await response.json();
    } catch (error) {
        console.error('Lỗi cập nhật:', error);
    }
}

// PATCH - Cập nhật một phần
async function patchUser(id, partialData) {
    try {
        const response = await fetch(
            `https://api.example.com/users/${id}`,
            {
                method: 'PATCH',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(partialData)
            }
        );
        return await response.json();
    } catch (error) {
        console.error('Lỗi cập nhật:', error);
    }
}

DELETE Request#

async function deleteUser(id) {
    try {
        const response = await fetch(
            `https://api.example.com/users/${id}`,
            {
                method: 'DELETE'
            }
        );
        
        if (response.ok) {
            console.log('Người dùng đã xóa');
        } else {
            throw new Error('Không thể xóa');
        }
    } catch (error) {
        console.error('Lỗi:', error);
    }
}

Headers tùy chỉnh#

const headers = new Headers({
    'Content-Type': 'application/json',
    'Authorization': 'Bearer token123',
    'X-Custom-Header': 'value'
});

fetch('https://api.example.com/data', {
    method: 'GET',
    headers: headers
})
.then(r => r.json())
.then(data => console.log(data));

Upload File#

async function uploadFile(file) {
    const formData = new FormData();
    formData.append('file', file);
    formData.append('description', 'File tải lên');
    
    try {
        const response = await fetch(
            'https://api.example.com/upload',
            {
                method: 'POST',
                body: formData
                // Không set Content-Type header
            }
        );
        return await response.json();
    } catch (error) {
        console.error('Lỗi upload:', error);
    }
}

// Sử dụng
const inputFile = document.getElementById('file-input');
inputFile.addEventListener('change', (e) => {
    uploadFile(e.target.files[0]);
});

Abort Request#

const controller = new AbortController();

setTimeout(() => {
    controller.abort(); // Hủy request sau 5 giây
}, 5000);

fetch('https://api.example.com/slow-endpoint', {
    signal: controller.signal
})
.then(r => r.json())
.catch(error => {
    if (error.name === 'AbortError') {
        console.log('Request bị hủy');
    }
});

Response Types#

async function handleResponses() {
    const response = await fetch('https://api.example.com/data');
    
    // JSON
    const json = await response.json();
    
    // Text
    const text = await response.text();
    
    // Blob (cho file)
    const blob = await response.blob();
    
    // ArrayBuffer
    const buffer = await response.arrayBuffer();
}

Error Handling#

async function robustFetch(url, options = {}) {
    const timeout = options.timeout || 5000;
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), timeout);
    
    try {
        const response = await fetch(url, {
            ...options,
            signal: controller.signal
        });
        
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        
        return await response.json();
        
    } catch (error) {
        if (error.name === 'AbortError') {
            throw new Error('Request timeout');
        }
        throw error;
    } finally {
        clearTimeout(timeoutId);
    }
}

Kết luận#

Fetch API là công cụ mạnh mẽ cho giao tiếp HTTP trong JavaScript. Nó cung cấp cách hiện đại và linh hoạt để tương tác với server.
