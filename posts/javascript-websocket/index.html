WebSocket - Giao tiếp thời gian thực với JavaScript

Tìm hiểu về WebSocket và cách sử dụng nó để xây dựng các ứng dụng giao tiếp thời gian thực.

WebSocket là gì?#

WebSocket cung cấp một kênh giao tiếp hai chiều qua một kết nối TCP duy nhất, lý tưởng cho các ứng dụng thời gian thực.

Server WebSocket với Node.js#

const WebSocket = require('ws');
const http = require('http');

const server = http.createServer();
const wss = new WebSocket.Server({ server });

// Quản lý kết nối
wss.on('connection', (ws) => {
    console.log('Client kết nối');
    
    // Nhận tin nhắn
    ws.on('message', (message) => {
        console.log('Nhận được:', message);
        
        // Gửi lại cho tất cả client
        wss.clients.forEach((client) => {
            if (client.readyState === WebSocket.OPEN) {
                client.send(`Broadcast: ${message}`);
            }
        });
    });
    
    // Khi client ngắt kết nối
    ws.on('close', () => {
        console.log('Client ngắt kết nối');
    });
    
    // Lỗi xử lý
    ws.on('error', (error) => {
        console.error('Lỗi WebSocket:', error);
    });
});

server.listen(8080, () => {
    console.log('WebSocket server chạy trên ws://localhost:8080');
});

Client WebSocket với JavaScript#

// Kết nối đến server
const ws = new WebSocket('ws://localhost:8080');

// Connection opened
ws.addEventListener('open', (event) => {
    console.log('Kết nối thành công');
    ws.send('Xin chào server!');
});

// Listen for messages
ws.addEventListener('message', (event) => {
    console.log('Tin nhắn từ server:', event.data);
    updateUI(event.data);
});

// Connection closed
ws.addEventListener('close', (event) => {
    console.log('Kết nối đóng');
});

// Connection error
ws.addEventListener('error', (event) => {
    console.error('Lỗi kết nối:', event);
});

// Gửi tin nhắn
function sendMessage(message) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(message);
    }
}

Ứng dụng Chat thời gian thực#

// Server - chat-server.js
const WebSocket = require('ws');
const http = require('http');

const server = http.createServer();
const wss = new WebSocket.Server({ server });

const clients = new Set();

wss.on('connection', (ws) => {
    clients.add(ws);
    
    // Thông báo user mới
    broadcast(JSON.stringify({
        type: 'user-joined',
        count: clients.size
    }));
    
    ws.on('message', (message) => {
        const data = JSON.parse(message);
        broadcast(JSON.stringify({
            type: 'message',
            text: data.text,
            timestamp: new Date()
        }));
    });
    
    ws.on('close', () => {
        clients.delete(ws);
        broadcast(JSON.stringify({
            type: 'user-left',
            count: clients.size
        }));
    });
});

function broadcast(message) {
    clients.forEach((client) => {
        if (client.readyState === WebSocket.OPEN) {
            client.send(message);
        }
    });
}

server.listen(8080);

Best Practices#

  1. Reconnection logic: Tự động kết nối lại khi bị ngắt
  2. Heartbeat: Gửi ping định kỳ để giữ kết nối
  3. Message validation: Kiểm tra dữ liệu đầu vào
  4. Resource cleanup: Đóng kết nối đúng cách

Kết luận#

WebSocket là công nghệ mạnh mẽ cho các ứng dụng yêu cầu giao tiếp thời gian thực như chat, notification, và gaming.
