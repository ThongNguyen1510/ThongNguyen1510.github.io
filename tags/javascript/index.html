Căn bản về JavaScript Async/Await cho lập trình mạng

Giới thiệu Async/Await Async/Await là cách hiện đại để xử lý các hoạt động bất đồng bộ trong JavaScript, đặc biệt hữu ích cho lập trình mạng. Callbacks vs Promises vs Async/Await Callback Hell function fetchData(url, callback) { fetch(url, (error, data) => { if (error) { fetch(url + '/retry', (retryError, retryData) => { if (retryError) callback(retryError); else callback(null, retryData); }); } else { callback(null, data); } }); } Với Async/Await async function fetchDataWithRetry(url) { try { const response = await fetch(url); return await response.json(); } catch (error) { console.log('Retry fetch...'); const retryResponse = await fetch(url + '/retry'); return await retryResponse.json(); } } Ví dụ thực tế async function getNetworkData() { try { // Gọi API const response = await fetch('https://api.example.com/data'); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const data = await response.json(); console.log('Dữ liệu:', data); return data; } catch (error) { console.error('Lỗi:', error); } } Xử lý nhiều yêu cầu // Serial - Lần lượt async function fetchSerially() { const data1 = await fetch('/api/data1').then(r => r.json()); const data2 = await fetch('/api/data2').then(r => r.json()); return [data1, data2]; } // Parallel - Song song async function fetchParallel() { const [response1, response2] = await Promise.all([ fetch('/api/data1'), fetch('/api/data2') ]); const [data1, data2] = await Promise.all([ response1.json(), response2.json() ]); return [data1, data2]; } Timeout xử lý function withTimeout(promise, ms) { return Promise.race([ promise, new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), ms) ) ]); } async function fetchWithTimeout(url, timeoutMs) { try { const response = await withTimeout(fetch(url), timeoutMs); return await response.json(); } catch (error) { console.error('Lỗi:', error); } } Kết luận Async/Await làm cho mã bất đồng bộ dễ đọc và dễ bảo trì hơn, rất quan trọng cho lập trình mạng hiệu quả.

Xây dựng Server Node.js với Express

Giới thiệu Express Express là một framework Node.js nhẹ nhàng và linh hoạt để xây dựng các ứng dụng web và API. Thiết lập dự án npm init -y npm install express Tạo Server cơ bản const express = require('express'); const app = express(); const PORT = 3000; // Middleware app.use(express.json()); // Routes app.get('/', (req, res) => { res.send('Chào mừng đến Server!'); }); app.get('/api/data', (req, res) => { res.json({ message: 'Đây là dữ liệu API', timestamp: new Date() }); }); app.post('/api/message', (req, res) => { const { text } = req.body; res.json({ received: text, status: 'success' }); }); // Error Handler app.use((err, req, res, next) => { console.error(err.stack); res.status(500).json({ error: 'Có lỗi xảy ra' }); }); // Start Server app.listen(PORT, () => { console.log(`Server chạy trên http://localhost:${PORT}`); }); Routing nâng cao // Route parameters app.get('/api/users/:id', (req, res) => { const userId = req.params.id; res.json({ userId: userId }); }); // Query parameters app.get('/api/search', (req, res) => { const query = req.query.q; res.json({ results: query }); }); // Multiple routes app.route('/api/posts') .get((req, res) => { res.json([ { id: 1, title: 'Post 1' }, { id: 2, title: 'Post 2' } ]); }) .post((req, res) => { res.status(201).json({ message: 'Post tạo thành công' }); }); Middleware // Custom middleware const logger = (req, res, next) => { console.log(`${req.method} ${req.path}`); next(); }; app.use(logger); // Authentication middleware const authenticate = (req, res, next) => { const token = req.headers.authorization; if (!token) { return res.status(401).json({ error: 'Chưa xác thực' }); } next(); }; app.get('/api/protected', authenticate, (req, res) => { res.json({ data: 'Dữ liệu được bảo vệ' }); }); CORS để cho phép cross-origin requests const cors = require('cors'); app.use(cors()); // Hoặc cấu hình chi tiết app.use(cors({ origin: 'http://localhost:3001', credentials: true })); Kết luận Express giúp xây dựng server mạng nhanh chóng và có cấu trúc tốt. Nó rất phù hợp cho các API REST và ứng dụng web.

WebSocket - Giao tiếp thời gian thực với JavaScript

WebSocket là gì? WebSocket cung cấp một kênh giao tiếp hai chiều qua một kết nối TCP duy nhất, lý tưởng cho các ứng dụng thời gian thực. WebSocket vs HTTP Tính chất HTTP WebSocket Hướng Một chiều Hai chiều Độ trễ Cao Thấp Overhead Cao Thấp Real-time Không Có Server WebSocket với Node.js const WebSocket = require('ws'); const http = require('http'); const server = http.createServer(); const wss = new WebSocket.Server({ server }); // Quản lý kết nối wss.on('connection', (ws) => { console.log('Client kết nối'); // Nhận tin nhắn ws.on('message', (message) => { console.log('Nhận được:', message); // Gửi lại cho tất cả client wss.clients.forEach((client) => { if (client.readyState === WebSocket.OPEN) { client.send(`Broadcast: ${message}`); } }); }); // Khi client ngắt kết nối ws.on('close', () => { console.log('Client ngắt kết nối'); }); // Lỗi xử lý ws.on('error', (error) => { console.error('Lỗi WebSocket:', error); }); }); server.listen(8080, () => { console.log('WebSocket server chạy trên ws://localhost:8080'); }); Client WebSocket với JavaScript // Kết nối đến server const ws = new WebSocket('ws://localhost:8080'); // Connection opened ws.addEventListener('open', (event) => { console.log('Kết nối thành công'); ws.send('Xin chào server!'); }); // Listen for messages ws.addEventListener('message', (event) => { console.log('Tin nhắn từ server:', event.data); updateUI(event.data); }); // Connection closed ws.addEventListener('close', (event) => { console.log('Kết nối đóng'); }); // Connection error ws.addEventListener('error', (event) => { console.error('Lỗi kết nối:', event); }); // Gửi tin nhắn function sendMessage(message) { if (ws.readyState === WebSocket.OPEN) { ws.send(message); } } Ứng dụng Chat thời gian thực // Server - chat-server.js const WebSocket = require('ws'); const http = require('http'); const server = http.createServer(); const wss = new WebSocket.Server({ server }); const clients = new Set(); wss.on('connection', (ws) => { clients.add(ws); // Thông báo user mới broadcast(JSON.stringify({ type: 'user-joined', count: clients.size })); ws.on('message', (message) => { const data = JSON.parse(message); broadcast(JSON.stringify({ type: 'message', text: data.text, timestamp: new Date() })); }); ws.on('close', () => { clients.delete(ws); broadcast(JSON.stringify({ type: 'user-left', count: clients.size })); }); }); function broadcast(message) { clients.forEach((client) => { if (client.readyState === WebSocket.OPEN) { client.send(message); } }); } server.listen(8080); Best Practices Reconnection logic: Tự động kết nối lại khi bị ngắt Heartbeat: Gửi ping định kỳ để giữ kết nối Message validation: Kiểm tra dữ liệu đầu vào Resource cleanup: Đóng kết nối đúng cách Kết luận WebSocket là công nghệ mạnh mẽ cho các ứng dụng yêu cầu giao tiếp thời gian thực như chat, notification, và gaming.

Fetch API và HTTP Requests trong JavaScript

Giới thiệu Fetch API Fetch API cung cấp một cách hiện đại để gọi các HTTP requests từ JavaScript, thay thế XMLHttpRequest. GET Request // Đơn giản fetch('https://api.example.com/users') .then(response => response.json()) .then(data => console.log(data)) .catch(error => console.error('Lỗi:', error)); // Với Async/Await async function getUsers() { try { const response = await fetch('https://api.example.com/users'); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const data = await response.json(); console.log(data); } catch (error) { console.error('Lỗi:', error); } } getUsers(); POST Request async function createUser(user) { try { const response = await fetch('https://api.example.com/users', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify(user) }); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const result = await response.json(); console.log('Người dùng tạo:', result); } catch (error) { console.error('Lỗi:', error); } } createUser({ name: 'Nguyễn Văn A', email: 'a@example.com' }); PUT/PATCH Request // PUT - Cập nhật toàn bộ async function updateUser(id, updatedData) { try { const response = await fetch( `https://api.example.com/users/${id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify(updatedData) } ); return await response.json(); } catch (error) { console.error('Lỗi cập nhật:', error); } } // PATCH - Cập nhật một phần async function patchUser(id, partialData) { try { const response = await fetch( `https://api.example.com/users/${id}`, { method: 'PATCH', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify(partialData) } ); return await response.json(); } catch (error) { console.error('Lỗi cập nhật:', error); } } DELETE Request async function deleteUser(id) { try { const response = await fetch( `https://api.example.com/users/${id}`, { method: 'DELETE' } ); if (response.ok) { console.log('Người dùng đã xóa'); } else { throw new Error('Không thể xóa'); } } catch (error) { console.error('Lỗi:', error); } } Headers tùy chỉnh const headers = new Headers({ 'Content-Type': 'application/json', 'Authorization': 'Bearer token123', 'X-Custom-Header': 'value' }); fetch('https://api.example.com/data', { method: 'GET', headers: headers }) .then(r => r.json()) .then(data => console.log(data)); Upload File async function uploadFile(file) { const formData = new FormData(); formData.append('file', file); formData.append('description', 'File tải lên'); try { const response = await fetch( 'https://api.example.com/upload', { method: 'POST', body: formData // Không set Content-Type header } ); return await response.json(); } catch (error) { console.error('Lỗi upload:', error); } } // Sử dụng const inputFile = document.getElementById('file-input'); inputFile.addEventListener('change', (e) => { uploadFile(e.target.files[0]); }); Abort Request const controller = new AbortController(); setTimeout(() => { controller.abort(); // Hủy request sau 5 giây }, 5000); fetch('https://api.example.com/slow-endpoint', { signal: controller.signal }) .then(r => r.json()) .catch(error => { if (error.name === 'AbortError') { console.log('Request bị hủy'); } }); Response Types async function handleResponses() { const response = await fetch('https://api.example.com/data'); // JSON const json = await response.json(); // Text const text = await response.text(); // Blob (cho file) const blob = await response.blob(); // ArrayBuffer const buffer = await response.arrayBuffer(); } Error Handling async function robustFetch(url, options = {}) { const timeout = options.timeout || 5000; const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), timeout); try { const response = await fetch(url, { ...options, signal: controller.signal }); if (!response.ok) { throw new Error(`HTTP ${response.status}: ${response.statusText}`); } return await response.json(); } catch (error) { if (error.name === 'AbortError') { throw new Error('Request timeout'); } throw error; } finally { clearTimeout(timeoutId); } } Kết luận Fetch API là công cụ mạnh mẽ cho giao tiếp HTTP trong JavaScript. Nó cung cấp cách hiện đại và linh hoạt để tương tác với server.
