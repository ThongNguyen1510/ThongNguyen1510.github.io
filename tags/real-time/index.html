WebSocket - Giao tiếp thời gian thực với JavaScript

WebSocket là gì? WebSocket cung cấp một kênh giao tiếp hai chiều qua một kết nối TCP duy nhất, lý tưởng cho các ứng dụng thời gian thực. const server = http.createServer(); const wss = new WebSocket.Server({ server }); // Quản lý kết nối wss.on('connection', (ws) => { console.log('Client kết nối'); // Nhận tin nhắn ws.on('message', (message) => { console.log('Nhận được:', message); // Gửi lại cho tất cả client wss.clients.forEach((client) => { if (client.readyState === WebSocket.OPEN) { client.send(`Broadcast: ${message}`); } }); }); // Khi client ngắt kết nối ws.on('close', () => { console.log('Client ngắt kết nối'); }); // Lỗi xử lý ws.on('error', (error) => { console.error('Lỗi WebSocket:', error); }); }); server.listen(8080, () => { console.log('WebSocket server chạy trên ws://localhost:8080'); }); Client WebSocket với JavaScript // Kết nối đến server const ws = new WebSocket('ws://localhost:8080');
